/* 1. OLAP_ORACLE_01_TOP_MEDICATIONS
   Afișează topul medicamentelor după ratingul mediu */
/* http://localhost:9990/DSA-SparkSQL-Service/rest/view/OLAP_ORACLE_01_TOP_MEDICATIONS */


 
CREATE OR REPLACE VIEW OLAP_ORACLE_01_TOP_MEDICATIONS AS
SELECT 
  m.name AS medication_name,
  ROUND(AVG(r.rating), 2) AS avg_rating,
  RANK() OVER (ORDER BY AVG(r.rating) DESC) AS ranking
FROM medication_review_view r
JOIN medication_view m ON r.medicationId = m.medicationId
GROUP BY m.name;

SELECT * FROM OLAP_ORACLE_01_TOP_MEDICATIONS;



/* 2. OLAP_POSTGRES_01_DISEASE_CUBE
   Distribuția pacienților pe gen, vârstă și boală (CUBE) */
/* http://localhost:9990/DSA-SparkSQL-Service/rest/view/OLAP_POSTGRES_01_DISEASE_CUBE */


-- OLAP_POSTGRES_01_DISEASE_CUBE
CREATE OR REPLACE VIEW OLAP_POSTGRES_01_DISEASE_CUBE AS
SELECT 
  NVL(p.gender, 'necunoscut') AS gender,
  NVL(
    CASE 
      WHEN p.age < 18 THEN '0-17'
      WHEN p.age BETWEEN 18 AND 35 THEN '18-35'
      WHEN p.age BETWEEN 36 AND 60 THEN '36-60'
      ELSE '60+'
    END, 'necunoscut') AS age_group,
  NVL(d.disease, 'necunoscut') AS disease,
  COUNT(*) AS nr_pacienti
FROM patient_view p
JOIN diagnosis_view d ON p.patientId = d.patientId
GROUP BY CUBE(
  p.gender,
  CASE 
    WHEN p.age < 18 THEN '0-17'
    WHEN p.age BETWEEN 18 AND 35 THEN '18-35'
    WHEN p.age BETWEEN 36 AND 60 THEN '36-60'
    ELSE '60+' 
  END,
  d.disease
);

SELECT * FROM OLAP_POSTGRES_01_DISEASE_CUBE;

/* http://localhost:9990/DSA-SparkSQL-Service/rest/view/OLAP_COMBINED_01_MEDICATION_COST_RATING */

/* 
. OLAP_COMBINED_09_AVG_RATING_BY_DISEASE_PIVOT
   Descriere: Pivot cu rating mediu per boală pentru fiecare medicament.
   Reformulare cu AVG(CASE WHEN ...) în loc de PIVOT.
   URL: http://localhost:9990/DSA-SparkSQL-Service/rest/view/OLAP_COMBINED_09_AVG_RATING_BY_DISEASE_PIVOT
*/

CREATE OR REPLACE VIEW OLAP_COMBINED_09_AVG_RATING_BY_DISEASE_PIVOT AS
SELECT 
  m.name AS medication_name,
  ROUND(AVG(CASE WHEN LOWER(m.disease) = 'diabetes' THEN r.rating END), 2) AS diabetes,
  ROUND(AVG(CASE WHEN LOWER(m.disease) = 'hypertension' THEN r.rating END), 2) AS hypertension,
  ROUND(AVG(CASE WHEN LOWER(m.disease) = 'cancer' THEN r.rating END), 2) AS cancer,
  ROUND(AVG(CASE WHEN LOWER(m.disease) = 'asthma' THEN r.rating END), 2) AS asthma,
  ROUND(AVG(CASE WHEN LOWER(m.disease) = 'arthritis' THEN r.rating END), 2) AS arthritis,
  ROUND(AVG(CASE WHEN LOWER(m.disease) = 'obesity' THEN r.rating END), 2) AS obesity,
  ROUND(AVG(CASE WHEN LOWER(m.disease) = 'heart disease' THEN r.rating END), 2) AS heart_disease,
  ROUND(AVG(CASE WHEN LOWER(m.disease) = 'depression' THEN r.rating END), 2) AS depression,
  ROUND(AVG(CASE WHEN LOWER(m.disease) = 'covid-19' THEN r.rating END), 2) AS covid,
  ROUND(AVG(CASE WHEN LOWER(m.disease) = 'flu' THEN r.rating END), 2) AS flu
FROM medication_view m
JOIN medication_review_view r ON m.medicationId = r.medicationId
GROUP BY m.name;

select * from OLAP_COMBINED_09_AVG_RATING_BY_DISEASE_PIVOT;

/*
OLAP_COMBINED_10_COSTS_ROLLUP_GROUPING
   Descriere: Media acoperirii de asigurare pe boală și tip, cu etichete pentru subtotaluri și total.
   Reformulare fără funcția GROUPING().
   URL: http://localhost:9990/DSA-SparkSQL-Service/rest/view/OLAP_COMBINED_10_COSTS_ROLLUP_GROUPING
*/

CREATE OR REPLACE VIEW OLAP_COMBINED_10_COSTS_ROLLUP_GROUPING AS
SELECT
  COALESCE(i.insuranceType, 'TOTAL') AS insurance_type,
  COALESCE(i.disease, 'TOTAL') AS disease,
  ROUND(AVG(i.coveragePercentage), 2) AS avg_coverage,
  CASE 
    WHEN i.insuranceType IS NULL AND i.disease IS NOT NULL THEN 'Subtotal pe boala'
    WHEN i.insuranceType IS NOT NULL AND i.disease IS NULL THEN 'Subtotal pe tip'
    WHEN i.insuranceType IS NULL AND i.disease IS NULL THEN 'TOTAL GENERAL'
    ELSE 'Detaliu'
  END AS rollup_label
FROM insurance_coverage_view i
GROUP BY ROLLUP(i.insuranceType, i.disease)
ORDER BY insurance_type, disease;

select * from OLAP_COMBINED_10_COSTS_ROLLUP_GROUPING;

/* 
	 OLAP_UNPIVOT_PATIENT_DETAILS
   Descriere: Transformă detalii despre pacienți (gen, vârstă, categorie, severitate) în format vertical.
   Reformulare sigură fără duplicări, folosind arrays și explode().
   URL: http://localhost:9990/DSA-SparkSQL-Service/rest/view/OLAP_UNPIVOT_PATIENT_DETAILS
*/

CREATE OR REPLACE VIEW OLAP_UNPIVOT_PATIENT_DETAILS AS
SELECT 
  patient_id,
  detail_type,
  detail_value
FROM (
  SELECT 
    p.patientId AS patient_id,
    array('Gender', 'Age', 'Category', 'Severity') AS detail_type_array,
    array(p.gender, CAST(p.age AS STRING), d.category, d.severity) AS detail_value_array
  FROM patient_view p
  JOIN diagnosis_view dx ON p.patientId = dx.patientId
  JOIN disease_view d ON LOWER(dx.disease) = LOWER(d.name)
) AS base
LATERAL VIEW posexplode(detail_type_array) dt AS idx, detail_type
LATERAL VIEW posexplode(detail_value_array) dv AS idx2, detail_value
WHERE dt.idx = dv.idx2;

select * from OLAP_UNPIVOT_PATIENT_DETAILS;

/* 14. OLAP_COMBINED_15_DISEASES_BY_DATE_PARTS
    Cazuri pe an, lună, zi și boală (ROLLUP) */
/* http://localhost:9990/DSA-SparkSQL-Service/rest/view/OLAP_COMBINED_15_DISEASES_BY_DATE_PARTS */



CREATE OR REPLACE VIEW OLAP_COMBINED_15_DISEASES_BY_DATE_PARTS AS

  EXTRACT(YEAR FROM dateDiagnosed) AS diagnosis_year,
  EXTRACT(MONTH FROM dateDiagnosed) AS diagnosis_month,

/* 15. OLAP_TREATMENT_COST_EVOLUTION
    Evoluția costului tratamentului în timp (LAG, LEAD) */
/* http://localhost:9990/DSA-SparkSQL-Service/rest/view/OLAP_TREATMENT_COST_EVOLUTION */


CREATE OR REPLACE VIEW OLAP_TREATMENT_COST_EVOLUTION AS
SELECT 
  d.disease,
  d.dateDiagnosed,
  ROUND(c.averageCostEur, 2) AS current_cost,
  LAG(c.averageCostEur) OVER (PARTITION BY d.disease ORDER BY d.dateDiagnosed) AS prev_cost,
  LEAD(c.averageCostEur) OVER (PARTITION BY d.disease ORDER BY d.dateDiagnosed) AS next_cost,
  FIRST_VALUE(c.averageCostEur) OVER (PARTITION BY d.disease ORDER BY d.dateDiagnosed) AS first_cost,
  LAST_VALUE(c.averageCostEur) OVER (
    PARTITION BY d.disease ORDER BY d.dateDiagnosed
    ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING
  ) AS last_cost
FROM diagnosis_view d
JOIN treatment_costs_view c ON LOWER(d.disease) = LOWER(c.disease);

select * from OLAP_TREATMENT_COST_EVOLUTION;

/* 17. OLAP_TREATMENT_RATING_TREND
    Tendința ratingurilor în timp pentru tratamente */
/* http://localhost:9990/DSA-SparkSQL-Service/rest/view/OLAP_TREATMENT_RATING_TREND */
CREATE OR REPLACE VIEW OLAP_TREATMENT_RATING_TREND AS
SELECT 
  t.treatmentId,
  d.disease,
  r.rating,
  d.dateDiagnosed,
  LAG(r.rating) OVER (PARTITION BY d.disease ORDER BY d.dateDiagnosed) AS prev_rating,
  r.rating - LAG(r.rating) OVER (PARTITION BY d.disease ORDER BY d.dateDiagnosed) AS rating_diff
FROM diagnosis_view d
JOIN treatment_view t ON d.diagnosisId = t.diagnosisId
JOIN medication_view m ON t.medicationId = m.medicationId
JOIN medication_review_view r ON m.medicationId = r.medicationId;

select * from OLAP_TREATMENT_RATING_TREND;
